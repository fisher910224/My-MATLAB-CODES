    methods (Access = public)
        function loadSettings(app, cfg)
            if isempty(cfg); return; end

            % 每個欄位都要確保「欄位存在」且「不是空的」才填入
            if isfield(cfg, 't1_suspect') && ~isempty(cfg.t1_suspect)
                app.T1SuspectEditField.Value = cfg.t1_suspect; 
            end
            if isfield(cfg, 't1_fail') && ~isempty(cfg.t1_fail)
                app.T1FailEditField.Value = cfg.t1_fail; 
            end

            if isfield(cfg, 't4_sensor_range') && ~isempty(cfg.t4_sensor_range)
                app.T4LowerlimitEditField.Value = cfg.t4_sensor_range(1); 
                app.T4UpperlimitEditField.Value = cfg.t4_sensor_range(2); 
            end
            if isfield(cfg, 't4_user_range') && ~isempty(cfg.t4_user_range)
                app.T4UserLowerlimitEditField.Value = cfg.t4_user_range(1); 
                app.T4UserUpperlimitEditField.Value = cfg.t4_user_range(2); 
            end

            if isfield(cfg, 't5_sensor_range') && isfield(cfg, 't5_user_range') && ~isempty(cfg.t5_sensor_range) && ~isempty(cfg.t5_user_range)
                app.UITable.Data = [cfg.t5_sensor_range, cfg.t5_user_range];
            end

            if isfield(cfg, 't6_threshold_low') && ~isempty(cfg.t6_threshold_low)
                app.T6LowerlimitEditField.Value = cfg.t6_threshold_low;
            end
            if isfield(cfg, 't6_threshold_high') && ~isempty(cfg.t6_threshold_high)
                app.T6UpperlimitEditField.Value = cfg.t6_threshold_high;
            end

            if isfield(cfg, 't7_n_dev') && ~isempty(cfg.t7_n_dev)
                app.T7ChangethresholdstdEditField.Value = cfg.t7_n_dev;
            end
            if isfield(cfg, 't7_tim_dev_hours') && ~isempty(cfg.t7_tim_dev_hours)
                app.T7StepsEditField.Value = cfg.t7_tim_dev_hours;
            end

            if isfield(cfg, 't8_rep_suspect') && ~isempty(cfg.t8_rep_suspect)
                app.T8SuspectEditField.Value = cfg.t8_rep_suspect;
            end
            if isfield(cfg, 't8_rep_fail') && ~isempty(cfg.t8_rep_fail)
                app.T8FailEditField.Value = cfg.t8_rep_fail;
            end
            if isfield(cfg, 't8_esp_val') && ~isempty(cfg.t8_esp_val)
                app.T8ChangerateEditField.Value = cfg.t8_esp_val;
            end

            if isfield(cfg, 't10_min_var_warn') && ~isempty(cfg.t10_min_var_warn)
                app.T10WarningthresholdEditField.Value = cfg.t10_min_var_warn;
            end
            if isfield(cfg, 't10_min_var_fail') && ~isempty(cfg.t10_min_var_fail)
                app.T10FailthresholdEditField.Value = cfg.t10_min_var_fail;
            end
            if isfield(cfg, 't10_n_tst') && ~isempty(cfg.t10_n_tst)
                app.T10TimestepsEditField.Value = cfg.t10_n_tst;
            end
            
        end
    end

        % Code that executes after component creation
        function startupFcn(app)
            % 1. 建立 12 列 x 4 欄的預設矩陣
            sensor_default_vals = repmat([-15, 15], 12, 1);
            user_default_vals = [
                0 1.1; 0 1.1; 0 1.1; 0 1.1; 0 1.1; 0 1.2; ... % 1-6月
                0 1.2; 0 1.2; 0 1.2; 0 1.2; 0 1.2; 0 1.1      % 7-12月
            ];
            
            % 2. 手動用程式碼設定欄位與列名稱
            app.UITable.ColumnName = {'Sensor lower limit', 'Sensor upper limit', 'User lower limit', 'User upper limit'};
            app.UITable.RowName = {'Jan.', 'Feb.', 'Mar.', 'Apr.', 'May.', 'Jun.', ...
                                   'Jul.', 'Aug.', 'Sep.', 'Oct.', 'Nov.', 'Dec.'};
        
            % 3. 注入純數值矩陣
            app.UITable.Data = [sensor_default_vals, user_default_vals];
            app.UITable.ColumnFormat = {'bank', 'bank', 'bank', 'bank'};

            if nargin > 1 && ~isempty(initialSettings)
                if nargin > 1 && ~isempty(initialSettings)
                    app.loadSettings(initialSettings);
                end
            end

        end

        % Button pushed function: CancelButton
        function CancelButtonPushed(app, event)
            app.Results = []; % 代表取消，不回傳任何東西
            uiresume(app.QCpropertiesCurrentUIFigure);
            app.QCpropertiesCurrentUIFigure.Visible = 'off';
        end

        % Button pushed function: OKButton
        function OKButtonPushed(app, event)

            app.Results.t1_suspect = app.T1SuspectEditField.Value;
            app.Results.t1_fail = app.T1FailEditField.Value;

            app.Results.t4_sensor_range = [app.T4LowerlimitEditField.Value, app.T4UpperlimitEditField.Value];
            app.Results.t4_user_range = [app.T4LowerlimitEditField_2.Value, app.T4UpperlimitEditField_2.Value];

            app.Results.t5_sensor_range = double(app.UITable.Data(:, 1:2));
            app.Results.t5_user_range   = double(app.UITable.Data(:, 3:4));

            app.Results.t6_threshold_low = app.T6LowerlimitEditField.Value;
            app.Results.t6_threshold_high = app.T6UpperlimitEditField.Value;

            app.Results.t7_n_dev = app.T7ChangethresholdstdEditField.Value;
            app.Results.t7_tim_dev_hours = app.T7StepsEditField.Value;

            app.Results.t8_rep_suspect = app.T8SuspectEditField.Value;
            app.Results.t8_rep_fail = app.T8FailEditField.Value;
            app.Results.t8_esp_val = app.T8ChangerateEditField.Value;

            app.Results.t10_min_var_warn = app.T10WarningthresholdEditField.Value;
            app.Results.t10_min_var_fail = app.T10FailthresholdEditField.Value;
            app.Results.t10_n_tst = app.T10TimestepsEditField.Value;
            
            uiresume(app.QCpropertiesCurrentUIFigure);
            % delete(app.QCpropertiesCurrentUIFigure);
            app.QCpropertiesCurrentUIFigure.Visible = 'off'; 
        end
    end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = Properties_QC_curr

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.QCpropertiesCurrentUIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion

            % Delete UIFigure when app is deleted
            delete(app.QCpropertiesCurrentUIFigure)
        end
