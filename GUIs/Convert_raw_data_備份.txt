    function autoDetectItemType(app)
        inputpath = app.InputdataEditField.Value;
        
        % 檢查路徑是否為空
        if isempty(inputpath) || inputpath == "select input data..."
            return; 
        end

        inputfilename = app.Inputfile;
        
        % 注意：邏輯順序很重要！
        % 必須先判斷 "stemp" 再判斷 "temp"，否則 "stemp" 也包含 "temp" 字串，會被攔截。
        if contains(inputfilename, "潮位站_stemp", "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA tidal station: Sea Temperature";
            app.InputItem = "t_stemp";
        elseif contains(inputfilename, ["stemp", "sea"], "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Sea Temperature";
            app.InputItem = "stemp";
        elseif contains(inputfilename, "temp", "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Temperature";
            app.InputItem = "temp";
        elseif contains(inputfilename, "tide", "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA tidal station: Sea Level";
            app.InputItem = "tide";
        elseif contains(inputfilename, ["curr", "cur"], "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Current";
            app.InputItem = "curr";
        elseif contains(inputfilename, ["wave", ".wap"], "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Wave";
            % app.ItemtypeDropDown_3.Value = "AWAC & AQP: Wave(21 columns)";
            app.InputItem = "wave";
        elseif contains(inputfilename, "wind", "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Wind";
            app.InputItem = "wind";
        elseif contains(inputfilename, "pres", "IgnoreCase", true)
            app.ItemtypeDropDown.Value = "CWA buoy: Atmosphere Pressure";
            app.InputItem = "pres";
        end
    end

        function percent_table = Wind_Beaufort_scale_statistics(app, Beaufort_scale, Dir, file_path)
            % 1. 定義級數與方向
            % Beaufort 從 0 到 12 (或 17，視資料而定)
            max_beaufort = 17;
            b_levels = 0:max_beaufort; 
            
            % 方向 16 方位
            directions_str = ["N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE", ...
                              "S", "SSW", "SW", "WSW", "W", "WNW", "NW", "NNW"];
            
            % 2. 方向角前處理 (處理北風跨越 0/360 度的問題)
            % N 的區間是 348.75~360 與 0~11.25
            % 將 > 348.75 的度數減去 360，使其落在 -11.25 ~ 11.25 之間
            Dir(Dir >= 348.75) = Dir(Dir >= 348.75) - 360;
            dir_edges = -11.25 : 22.5 : 348.75; % 17 個點構成 16 個區間
            
            % 3. 建立統計矩陣 (Rows: Beaufort 0~17 + Total, Cols: 16 Dir + Total)
            num_b = length(b_levels);
            num_d = length(directions_str);
            count_table = zeros(num_b + 1, num_d + 1);
            
            % 4. 開始統計 (建議使用邏輯索引，效率較高)
            for r = 1:num_b
                target_b = b_levels(r);
                for c = 1:num_d
                    % 找出符合該級數且在該方向區間內的資料
                    idx = (Beaufort_scale == target_b) & ...
                          (Dir > dir_edges(c)) & (Dir <= dir_edges(c+1));
                    count_table(r, c) = sum(idx);
                end
            end
            
            % 5. 計算邊際總和 (Marginal Sums)
            count_table(1:num_b, end) = sum(count_table(1:num_b, 1:num_d), 2); % 每列總和
            count_table(end, 1:num_d) = sum(count_table(1:num_b, 1:num_d), 1); % 每行總和
            total_samples = sum(count_table(1:num_b, end));
            count_table(end, end) = total_samples;
            
            % 6. 轉為百分比 (%)
            if total_samples > 0
                percent_table = (count_table ./ total_samples) * 100;
            else
                percent_table = count_table;
            end
            
            % 7. 寫入檔案
            fid = fopen(file_path, "w", "n", "UTF-8");
            if fid == -1, uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, "Cannot open file!", "Error"); end
            fwrite(fid, [239 187 191], 'uint8');      % 加入這一行寫入 BOM
            
            % 寫入標題 (補上一個空白對齊 Beaufort 欄位)
            fprintf(fid, "%s", "Bft/Dir");
            fprintf(fid, ",%s", directions_str);
            fprintf(fid, ",%s\n", "Total(%)");
            
            % 寫入各級數資料
            for r = 1:num_b + 1
                if r <= num_b
                    row_label = sprintf("%d", b_levels(r));
                else
                    row_label = "Total(%)";
                end
                
                fprintf(fid, "%s", row_label);
                % 寫入該列所有的百分比
                fprintf(fid, ",%.1f", percent_table(r, :));
                fprintf(fid, "\n");
            end
            
            fclose(fid);
            % fprintf("統計完成，存檔至: %s\n", file_path);
            
        end
end

        % Button pushed function: BrowseButton
        function browse_raw_data(app, event)
            % startPath = 'C:\Users\adm\Desktop'; 
            % selPath = uigetdir(startPath, 'Select input file folder...');
            selType = {'*.*',  'All Files'; '*.txt', '.txt'; '*.csv', '.csv'; '*.xlsx', '.xlsx'; };
            [filename, selPath, ~] = uigetfile(selType, 'Select input file folder...', "MultiSelect", "on", "");
            % selPath = uigetdir('', 'Select input file folder...');
% 
            if isequal(selPath, 0)
                figure(app.CWAOceanDataProcessingStatisticsToolv10UIFigure); 
                return;
            else
                app.InputdataEditField.Value = selPath;
                app.input_folder = selPath;
                if ischar(filename) 
                    app.Inputfile = {filename}; % 單選變 {'file.txt'}
                else
                    app.Inputfile = filename;   % 多選本來就是 cell
                end
                
                % --- 在這裡呼叫自動偵測 ---
                app.autoDetectItemType(); 
                
                figure(app.CWAOceanDataProcessingStatisticsToolv10UIFigure); 
            end

            
        end

        % Button pushed function: StartOutputFileButton
        function StartOutputFileButtonPushed(app, event)
            % app.main_folder = app.InputdataEditField.Value;
            
            
            try

            if isempty(app.input_folder) || ~isfolder(app.input_folder)
                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Empty input path!', 'Error')
                return;
            end

            if isempty(app.OutputfolderEditField.Value) || ~isfolder(app.OutputfolderEditField.Value)
                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Empty output path!', 'Error')
                return;
            end

            if app.InputItem == ""
                % 手動觸發一次對應邏輯，或者直接從下拉選單判斷
                ItemtypeDropDownValueChanged(app, event); 
            end


            % searchKey = app.InputItem;
            % file_name_txt = "*" + searchKey + ".txt";
            % file_name_csv = "*" + searchKey + ".csv";
            % file_name_txt = "*.txt";
            % file_name_csv = "*.csv";
            % dd_txt = dir(fullfile(app.Inputfile, file_name_txt));
            % dd_csv = dir(fullfile(app.Inputfile, file_name_csv));
            selectedFiles = app.Inputfile;
            if ischar(selectedFiles)
                selectedFiles = {selectedFiles};
            end
            isTxt = endsWith(selectedFiles, '.txt', 'IgnoreCase', true);
            isCsv = endsWith(selectedFiles, '.csv', 'IgnoreCase', true);

            dd_txt = selectedFiles(isTxt);
            dd_csv = selectedFiles(isCsv);

            if isempty(dd_txt) && isempty(dd_csv)
                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Cannot find file!', 'Error')
                return;
            elseif ~isempty(dd_txt) && isempty(dd_csv)
                dd = dd_txt;
                currentFormat = 'txt';
            elseif isempty(dd_txt) && ~isempty(dd_csv)
                dd = dd_csv;
                currentFormat = 'csv';
            else
                % select a file type
                selection = uiconfirm(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, ...
                    'There are .txt and .csv file, which one do you want to process?', ...
                    'Select a file type', ...
                    'CancelOption', 3, ... % 這行關鍵：按 X 或 Esc 會回傳 'Cancel'
                    'Options', {'.txt', '.csv', 'cancel'}, ...
                    'DefaultOption', 3, ...
                    'Icon', 'question');
                if strcmp(selection, '.txt')
                    dd = dd_txt;
                    currentFormat = 'txt';
                elseif strcmp(selection, '.csv')
                    dd = dd_csv;
                    currentFormat = 'csv';
                else 
                    return;
                end
            end
            

            switch currentFormat
                case 'txt'

                    numFiles = length(dd); % 總檔案數
    
                    % 1. 建立進度條對話框
                    d = uiprogressdlg(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Title', 'Processing...', ...
                        'Message', 'Preparing...', 'Cancelable', 'on');
                    data = struct();

                    userCancelled = false;

                    if d.CancelRequested
                        return;
                    end

                    switch app.InputItem
                        case 'temp'
                            
                            % ============= Temperature =============
                            opts = fixedWidthImportOptions('NumVariables', 4);
                            if ~isempty(app.StartReadingRaw)
                                opts.DataLines = app.StartReadingRaw;
                            else
                                opts.DataLines = 14;
                            end
                            opts.VariableNames = {'st', 'yyyymmddhhmi', 'Z', 'T'};
                            opts.VariableWidths = [7, 14, 2, 5]; 
                            opts.VariableTypes = {'string', 'string', 'double', 'double'};
                            opts.MissingRule = 'fill';
                            for ii = 1:length(dd)
                                if d.CancelRequested
                                    userCancelled = true;
                                    break; % 跳出 for 迴圈
                                end

                                df = readtable(fullfile(app.input_folder, app.Inputfile{ii}), opts, "ReadVariableNames", false);
                                data.temp(ii).ddatetime = datetime(string(df.yyyymmddhhmi), 'Inputformat', 'yyyyMMddHHmm');
                                data.temp(ii).dday = datenum(data.temp(ii).ddatetime);
                                data.temp(ii).temp = df.T;
                                d.Value = ii / numFiles; 
                                d.Message = sprintf('Processing %d/%d file: %s', ii, numFiles, app.Inputfile{ii});
                                
                            end

                            close(d);
                            if userCancelled
                                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Process cancelled by user.', 'Cancelled');
                                return; % 直接結束 StartButtonPushed 函式
                            end

                            % if d.CancelRequested
                            %     uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, '處理已取消', '訊息');
                            % else

                            % end

                        case 'stemp'
                            % ============= Sea Temperature =============
                            opts = fixedWidthImportOptions('NumVariables', 4);
                            if ~isempty(app.StartReadingRaw)
                                opts.DataLines = app.StartReadingRaw;

                            for ii = 1:length(dd)

                            if userCancelled
                                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Process cancelled by user.', 'Cancelled');

                        case 't_stemp'

        % Button pushed function: PropertiesButton_3
        function PropertiesButtonPushed(app, event)
            try
                % 1. 打包主程式當前的設定（第一次打開若為空，就傳空值 []）
                if ~isempty(app.wind_statistics_colormap)
                    currentSettings = struct(...
                        'Colormap', app.wind_statistics_colormap, ...
                        'IsFlipped', app.wind_statistics_colormap_IsFlipped, ...
                        'Format', app.wind_statistics_colormap_format, ...
                        'Dirplot_ylim', app.wind_DirPlot_ylim, ...
                        'WindSpeed_ylim', app.wind_WindSpeed_ylim, ...
                        'WindRoseColormap', app.WindRoseColormap, ...
                        'all_MaxFrequency', app.WindRose_all_MaxFrequency, ...
                        'all_nfreq', app.WindRose_all_nfreq, ...
                        'mon_MaxFrequency', app.WindRose_mon_MaxFrequency, ...
                        'mon_nfreq', app.WindRose_mon_nfreq);
                else
                    currentSettings = []; % 第一次打開，讓子視窗使用預設值
                end
        
                % 2. 關鍵！把當前設定傳進子視窗括號裡：
                settingsWin = Properties_statistics(currentSettings); 
        
                % 3. 等待使用者按 Ok / Cancel
                uiwait(settingsWin.UIFigure);
        
                % 4. 判斷使用者是否按了 Ok
                if isvalid(settingsWin)
                    if ~isempty(settingsWin.Results)
                        % 拿到最新修改的設定值
                        res = settingsWin.Results;
        
                        % 存回主 App 的各個變數中
                        app.wind_statistics_colormap          = res.Colormap;
                        app.wind_statistics_colormap_IsFlipped = res.IsFlipped;
                        app.wind_statistics_colormap_format    = res.Format;
                        app.wind_DirPlot_ylim                 = res.Dirplot_ylim;
                        app.wind_WindSpeed_ylim               = res.WindSpeed_ylim;
                        
                        app.WindRoseColormap                  = res.WindRoseColormap;
                        app.WindRose_all_MaxFrequency         = res.all_MaxFrequency;
                        app.WindRose_all_nfreq                = res.all_nfreq;
                        app.WindRose_mon_MaxFrequency         = res.mon_MaxFrequency;
                        app.WindRose_mon_nfreq                = res.mon_nfreq;
        
                        uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, 'Updated properties!', 'Success');
                    end
                    
                    % 不論按 Ok 還是 Cancel，等待結束後都把子視窗銷毀釋放記憶體
                    delete(settingsWin); 
                end
        
            catch ME
                errorMessage = sprintf('Coding error! \n\nReason: %s\n\nLocation:%s (Line: %d)', ...
                    ME.message, ME.stack(end).name, ME.stack(end).line);
                uialert(app.CWAOceanDataProcessingStatisticsToolv10UIFigure, errorMessage, 'System error', 'Icon', 'error');
            end
            
        end
