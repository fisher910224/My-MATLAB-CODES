        FrequecyintervalSpinner_2Label  matlab.ui.control.Label
        SeasonalMonthly0AutoLabel       matlab.ui.control.Label
        Allyear0AutoLabel               matlab.ui.control.Label
        RoseMaxFreqSpinnerLabel_2       matlab.ui.control.Label
        FrequecyintervalSpinner         matlab.ui.control.Spinner
        FrequecyintervalSpinnerLabel    matlab.ui.control.Label
        FrequecyintervalSpinner_2       matlab.ui.control.Spinner
        RoseMaxFreqSpinner              matlab.ui.control.Spinner
        RoseMaxFreqSpinner_2            matlab.ui.control.Spinner
        RoseMaxFreqSpinnerLabel         matlab.ui.control.Label
        PictureexportformatDropDown     matlab.ui.control.DropDown
        PictureexportformatLabel        matlab.ui.control.Label
        StasticsseasonalplotPanel       matlab.ui.container.Panel
        WindspeedprobYlimSpinner        matlab.ui.control.Spinner

    properties (Access = public)
        Results = struct('Colormap', 'gray', 'IsFlipped', true, 'Format', '.png', ...
            'Dirplot_ylim', 30, 'WindSpeed_ylim', 50, ...
            'all_MaxFrequency', 0, 'all_nfreq', 4, 'mon_MaxFrequency', 0, 'mon_nfreq', 5) % Description
    end

    methods (Access = public)
        function loadSettings(app, cfg)
            if isempty(cfg); return; end

            % 每個欄位都要確保「欄位存在」且「不是空的」才填入
            if isfield(cfg, 'Colormap') && ~isempty(cfg.Colormap)
                app.ColormapDropDown.Value = cfg.Colormap; 
            end
            if isfield(cfg, 'IsFlipped') && ~isempty(cfg.IsFlipped)
                app.FlipcoloruptodownCheckBox.Value = cfg.IsFlipped; 
            end
            if isfield(cfg, 'Format') && ~isempty(cfg.Format)
                app.PictureexportformatDropDown.Value = char(cfg.Format); 
            end
            if isfield(cfg, 'Dirplot_ylim') && ~isempty(cfg.Dirplot_ylim)
                app.DirectionprobYlimSpinner.Value = cfg.Dirplot_ylim; 
            end
            if isfield(cfg, 'WindSpeed_ylim') && ~isempty(cfg.WindSpeed_ylim)
                app.WindspeedprobYlimSpinner.Value = cfg.WindSpeed_ylim; 
            end
            if isfield(cfg, 'all_MaxFrequency') && ~isempty(cfg.all_MaxFrequency)
                app.RoseMaxFreqSpinner.Value = cfg.all_MaxFrequency; 
            end
            if isfield(cfg, 'all_nfreq') && ~isempty(cfg.all_nfreq)
                app.FrequecyintervalSpinner.Value = cfg.all_nfreq; 
            end

            if isfield(cfg, 'mon_MaxFrequency') && ~isempty(cfg.mon_MaxFrequency)
                app.RoseMaxFreqSpinner_2.Value = cfg.mon_MaxFrequency;
            else
                app.RoseMaxFreqSpinner_2.Value = 0;
            end
            
            if isfield(cfg, 'mon_nfreq') && ~isempty(cfg.mon_nfreq)
                app.FrequecyintervalSpinner_2.Value = cfg.mon_nfreq;
            else
                app.FrequecyintervalSpinner_2.Value = 0;
            end
        end
    end
    

    % Callbacks that handle component events
    methods (Access = private)

        % Code that executes after component creation
        function startupFcn(app, initialSettings)
            % Code that executes after component creation 
            if nargin > 1 && ~isempty(initialSettings)
                if nargin > 1 && ~isempty(initialSettings)
                    app.loadSettings(initialSettings);
                end
            end
        end

        % Button pushed function: OkButton
        function OkButtonPushed(app, event)
            app.Results.Colormap = app.ColormapDropDown.Value;
            app.Results.IsFlipped = app.FlipcoloruptodownCheckBox.Value;
            app.Results.Format = app.PictureexportformatDropDown.Value;

            app.Results.Dirplot_ylim = app.DirectionprobYlimSpinner.Value;
            app.Results.WindSpeed_ylim = app.WindspeedprobYlimSpinner.Value;

            app.Results.all_MaxFrequency = app.RoseMaxFreqSpinner.Value;
            app.Results.all_nfreq = app.FrequecyintervalSpinner.Value;
            if app.RoseMaxFreqSpinner_2.Value ~= 0
                app.Results.mon_MaxFrequency = app.RoseMaxFreqSpinner_2.Value;
            else
                app.Results.mon_MaxFrequency = [];
            end
            if app.FrequecyintervalSpinner_2.Value ~= 0
                app.Results.mon_nfreq = app.FrequecyintervalSpinner_2.Value;
            else
                app.Results.mon_nfreq = [];
            end

            uiresume(app.UIFigure);
            % delete(app.UIFigure);
            app.UIFigure.Visible = 'off'; 
        end

        % Button pushed function: CancelButton
        function CancelButtonPushed(app, event)
                app.Results = []; % 代表取消，不回傳任何東西
                uiresume(app.UIFigure);
                app.UIFigure.Visible = 'off';
        end
    end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = Properties_statistics(varargin)

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @(app)startupFcn(app, varargin{:}))

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
    end
