        function percent_table = Wind_Beaufort_scale_statistics(app, Beaufort_scale, Dir, file_path)
            % 1. 定義級數與方向
            % Beaufort 從 0 到 12 (或 17，視資料而定)
            max_beaufort = 17;
            b_levels = 0:max_beaufort; 
            
            % 方向 16 方位
            directions_str = ["N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE", ...
                              "S", "SSW", "SW", "WSW", "W", "WNW", "NW", "NNW"];
            
            % 2. 方向角前處理 (處理北風跨越 0/360 度的問題)
            % N 的區間是 348.75~360 與 0~11.25
            % 將 > 348.75 的度數減去 360，使其落在 -11.25 ~ 11.25 之間
            Dir(Dir >= 348.75) = Dir(Dir >= 348.75) - 360;
            dir_edges = -11.25 : 22.5 : 348.75; % 17 個點構成 16 個區間
            
            % 3. 建立統計矩陣 (Rows: Beaufort 0~17 + Total, Cols: 16 Dir + Total)
            num_b = length(b_levels);
            num_d = length(directions_str);
            count_table = zeros(num_b + 1, num_d + 1);
            
            % 4. 開始統計 (建議使用邏輯索引，效率較高)
            for r = 1:num_b
                target_b = b_levels(r);
                for c = 1:num_d
                    % 找出符合該級數且在該方向區間內的資料
                    idx = (Beaufort_scale == target_b) & ...
                          (Dir > dir_edges(c)) & (Dir <= dir_edges(c+1));
                    count_table(r, c) = sum(idx);
                end
            end
            
            % 5. 計算邊際總和 (Marginal Sums)
            count_table(1:num_b, end) = sum(count_table(1:num_b, 1:num_d), 2); % 每列總和
            count_table(end, 1:num_d) = sum(count_table(1:num_b, 1:num_d), 1); % 每行總和
            total_samples = sum(count_table(1:num_b, end));
            count_table(end, end) = total_samples;
            
            % 6. 轉為百分比 (%)
            if total_samples > 0
                percent_table = (count_table ./ total_samples) * 100;
            else
                percent_table = count_table;
            end
            
            % 7. 寫入檔案
            fid = fopen(file_path, "w", "n", "UTF-8");
            if fid == -1, uialert(app.UIFigure, "Cannot open file!", "Error"); end
            fwrite(fid, [239 187 191], 'uint8');      % 加入這一行寫入 BOM
            
            % 寫入標題 (補上一個空白對齊 Beaufort 欄位)
            fprintf(fid, "%s", "Bft/Dir");
            fprintf(fid, ",%s", directions_str);
            fprintf(fid, ",%s\n", "Total(%)");
            
            % 寫入各級數資料
            for r = 1:num_b + 1
                if r <= num_b
                    row_label = sprintf("%d", b_levels(r));
                else
                    row_label = "Total(%)";
                end
                
                fprintf(fid, "%s", row_label);
                % 寫入該列所有的百分比
                fprintf(fid, ",%.1f", percent_table(r, :));
                fprintf(fid, "\n");
            end
            
            fclose(fid);
            % fprintf("統計完成，存檔至: %s\n", file_path);
            
        end
end

        function BrowseButtonPushed(app, event)
            % startPath = 'C:\Users\adm\Desktop'; 
            % selPath = uigetdir(startPath, 'Select input file folder...');
            selType = {'*.*',  'All Files(*)'; '*.csv', '.csv(Current)'; '*.wap', '.wap(Wave)'};
            [filename, selPath, ~] = uigetfile(selType, 'Select input file folder...', "MultiSelect", "on", "");

            if isequal(selPath, 0)

                return;

                app.InputdataEditField.Value = selPath;
                app.input_folder = selPath;
                if ischar(filename) 
                    app.Inputfile = {filename}; % 單選變 {'file.txt'}
                else
                    app.Inputfile = filename;   % 多選本來就是 cell
                end
                
                % --- 在這裡呼叫自動偵測 ---
                app.autoDetectItemType(); 
                
                figure(app.UIFigure); 
            end
        end

        % Button pushed function: StartOutputFileButton
        function StartOutputFileButtonPushed(app, event)
            try

                if isempty(app.input_folder) || ~isfolder(app.input_folder)
                    uialert(app.UIFigure, 'Empty input path!', 'Error')
                    return;
                end
    
                if isempty(app.OutputfolderEditField.Value) || ~isfolder(app.OutputfolderEditField.Value)
                    uialert(app.UIFigure, 'Empty output path!', 'Error')
                    return;
                end
    
                if app.InputItem == ""
                    % 手動觸發一次對應邏輯，或者直接從下拉選單判斷
                    ItemtypeDropDownValueChanged(app, event); 
                end
    
                selectedFiles = app.Inputfile;
                if ischar(selectedFiles)
                    selectedFiles = {selectedFiles};
                end
    
                % 1. 建立進度條對話框
                d = uiprogressdlg(app.UIFigure, 'Title', 'Processing...', ...
                    'Message', 'Preparing...', 'Cancelable', 'on');
                data = struct();
                userCancelled = false;

                if d.CancelRequested
                    return;
                end

                switch app.InputItem
                    case 'curr'
                        % ============= Current =============
                        data = struct();
                        isCsv = endsWith(selectedFiles, '.csv', 'IgnoreCase', true);
                        dd_csv = selectedFiles(isCsv);
                        if isempty(dd_csv)
                            uialert(app.UIFigure, "CANNOT find .csv file!", "File NOT found.")
                            return
                        end
                        dd = dd_csv;
                        numFiles = length(dd);
                        data.curr = struct('ddatetime', [], 'dday', [], 'Cur', [], 'Dir', []);

                        for ii = 1:length(dd)
                            if d.CancelRequested
                                userCancelled = true;
                                break; % 跳出 for 迴圈
                            end

                            df = readtable(fullfile(app.input_folder, app.Inputfile{ii}), 'ReadVariableNames', false, 'Delimiter', ';');
                            % df = readmatrix(fullfile(app.main_folder, app.Inputfile{ii}), 'Delimiter', ' ');
                            if isempty(app.StartLine)
                                app.lowest_SL = mean(df.Var6, "omitmissing") - mean(diff(df.Var6));
                                ig = find(df.Var6 >= app.lowest_SL);
                            else
                                ig = app.StartLine:length(df);
                            end
                            

                            if numFiles >= 2
                                data.curr_sepa(ii).ddatetime = datetime(string(df.Var1(ig(1):ig(end))), 'Inputformat', 'yyyy/MM/dd HH:mm:ss');
                                data.curr_sepa(ii).dday = datenum(data.curr_sepa(ii).ddatetime);
                                data.curr_sepa(ii).Cur = table2array(df(ig(1):ig(end), 10:2:end));
                                data.curr_sepa(ii).Dir = table2array(df(ig(1):ig(end), 11:2:end));
                                data.curr.ddatetime = [data.curr.ddatetime; data.curr_sepa(ii).ddatetime];
                                data.curr.dday = [data.curr.dday; datenum(data.curr_sepa(ii).ddatetime)];
                                data.curr.Cur = [data.curr.Cur; data.curr_sepa(ii).Cur];
                                data.curr.Dir = [data.curr.Dir; data.curr_sepa(ii).Dir];
                            else
                                data.curr.ddatetime = datetime(string(df.Var1(ig(1):ig(end))), 'Inputformat', 'yyyy/MM/dd HH:mm:ss');
                                data.curr.dday = datenum(data.curr_sepa(ii).ddatetime);
                                data.curr.Cur = table2array(df(ig(1):ig(end), 10:2:end));
                                data.curr.Dir = table2array(df(ig(1):ig(end), 11:2:end));
                            end

                            d.Value = ii / numFiles; 
                            d.Message = sprintf('Processing %d/%d file: %s', ii, numFiles, app.Inputfile{ii});

                        data.curr.Cur = data.curr.Cur(sort_idx);

                        close(d);

        % Button pushed function: StartQCButton
        function StartQCButtonPushed(app, event)
            try

                if isempty(app.input_folder_2) || ~isfolder(app.input_folder_2)
                    uialert(app.UIFigure, 'Empty input path!', 'Error')
                    return;
                end
    
                if isempty(app.OutputfolderEditField_2.Value) || ~isfolder(app.OutputfolderEditField_2.Value)
                    uialert(app.UIFigure, 'Empty output path!', 'Error')
                    return;
                end
                
                selectedFile = app.SelectfileEditField.Value;
    
                % 1. 建立進度條對話框
                d = uiprogressdlg(app.UIFigure, 'Title', 'Processing...', ...
                    'Message', 'Preparing...', 'Cancelable', 'on');
                data = struct();
                userCancelled = false;

                if d.CancelRequested
                    return;
                end

                % app.QCItem = app.QCItemDropDown.Value;
                value = string(app.QCItemDropDown.Value);
                    % value = string(app.ItemtypeDropDown.Value);
                    
                switch value
                    case "Wave (Hs, Tp, Tmean, Dir)"
                        app.QCItem = "wave";
    
                    case "Current (Vel, Dir)"
                        app.QCItem = "curr";
    
                    case "Sea Level"
                        app.QCItem = "sl";
                end

                switch app.QCItem
                    case 'curr'
                        data.curr = struct();
                        
                        df = load(selectedFile);
                        if fieldnames(df) ~= app.QCItem
                            alert_str = sprintf("Selected wrong QC Item!\nSuggest Select QC item: %s.", string(fieldnames(df)));
                            uialert(app.UIFigure, alert_str, "Error"); 
                            return; 
                        end
                        filenames = fieldnames(df.((app.QCItem)));
                        keep_items = {'ddatetime', 'Cur', 'Dir'};
                        filenames_keep = filenames(matches(filenames, keep_items));
                        curr = struct();
                       
                        for ii = 1:length(df.((app.QCItem)))
                            % ff_name = filenames_keep{ii};
                            % 1. 先保留原始未補齊的時間序列
                            raw_time = df.curr(ii).ddatetime;
                            
                            % 2. 分別將 Cur 和 Dir 對齊到補齊後的完整網格
                            [new_time, curr(ii).Cur] = find_missing_data(raw_time, df.curr(ii).Cur);
                            [~,        curr(ii).Dir] = find_missing_data(raw_time, df.curr(ii).Dir);
                            
                            % 3. 最後再將 ddatetime 更新為補齊後的完整時間序列
                            curr(ii).ddatetime = new_time;
                        end

                        for ii = 1:length(curr)
                            if d.CancelRequested, userCancelled = true; break; end
                            d.Value = ii / length(curr); 
                            d.Message = sprintf('QC %d/%d file: %s', ii, length(curr), datestr(curr(ii).ddatetime(1), 'yyyy'));
                            if app.sr == 0
                                app.sr = mode(diff(curr(ii).ddatetime));
                                app.sr = hours(app.sr);
                            end
                            data.curr(ii).ddatetime = curr(ii).ddatetime;
                            opts_cell = namedargs2cell(app.QC_switch);
                            [Cur_flags, f] = QC_TEST(curr(ii).ddatetime, curr(ii).Cur, app.sr, app.CU_QCsettings, opts_cell{:});
                            data.curr(ii).Cur = curr(ii).Cur;
                            data.curr(ii).Cur(f >= 4) = NaN;
                            data.curr(ii).Cur_flags = Cur_flags;
    
                            % [Dir_flags, f] = QC_TEST(ddatetime, Dir, sr, CD_QCsettings, opts_cell{:});
                            [Dir_flags, f] = QC_TEST(curr(ii).ddatetime, curr(ii).Dir, app.sr, app.CD_QCsettings, "Test1", "on", "Test4", "on", "Test5", "on", ...
                                "Test6", "off", "Test7", "off", "Test8", "off", "Test10", "off");
                            data.curr(ii).Dir = curr(ii).Dir;
                            data.curr(ii).Dir(f >= 4) = NaN;
                            data.curr(ii).Dir_flags = Dir_flags;
                        end

                    case 'wave'
                        data.wave = struct();

                        df = load(selectedFile);
                        if fieldnames(df) ~= app.QCItem
                            alert_str = sprintf("Selected wrong QC Item!\nSuggest Select QC item: %s.", string(fieldnames(df)));
                            uialert(app.UIFigure, alert_str, "Error"); 
                            return; 
                        end
                        filenames = fieldnames(df.((app.QCItem)));
                        keep_items = {'ddatetime', 'Hs', 'Tmean', 'Tp', 'Dir'};
                        filenames_keep = filenames(matches(filenames, keep_items));
                        wave = struct();
                        % sr = app.sr;
                       
                        for ii = 1:length(df.((app.QCItem)))
                        % for ii = 1
                            % ff_name = filenames_keep{ii};
                            % 1. 先保留原始未補齊的時間序列
                            raw_time = df.wave(ii).ddatetime;

                            if app.sr == 0
                                app.sr = mode(diff(df.wave(ii).ddatetime));
                                app.sr = hours(app.sr);
                            else
                                app.sr = hours(app.sr);
                            end
                            
                            % 2. 分別將 Cur 和 Dir 對齊到補齊後的完整網格
                            [new_time, wave(ii).Hs] = find_missing_data(raw_time, df.wave(ii).Hs, app.sr);
                            [~,        wave(ii).Tmean] = find_missing_data(raw_time, df.wave(ii).Tmean, app.sr);
                            [~,        wave(ii).Tp] = find_missing_data(raw_time, df.wave(ii).Tp, app.sr);
                            [~,        wave(ii).Dir] = find_missing_data(raw_time, df.wave(ii).Dir, app.sr);
                            
                            % 3. 最後再將 ddatetime 更新為補齊後的完整時間序列
                            wave(ii).ddatetime = new_time;
                            wave(ii).raw_time = raw_time;
                        end

                        opts_cell = namedargs2cell(app.QC_switch);

                        for ii = 1:length(wave)

                            if d.CancelRequested, userCancelled = true; break; end

                            d.Value = ii / length(wave); 
                            d.Message = sprintf('QC %d/%d file: %s', ii, length(wave), datestr(wave(ii).ddatetime(1), 'yyyy'));
                            % sr = mode(diff(wave(ii).ddatetime));
                            % sr = hours(sr);
                            
                            data.wave(ii).ddatetime = wave(ii).ddatetime;
                            if app.sr == 0
                                app.sr = mode(diff(wave(ii).ddatetime));
                                app.sr = hours(app.sr);
                            end
                            

                            [Hs_flags, f] = QC_TEST(wave(ii).ddatetime, wave(ii).Hs, app.sr, app.WV_QCsettings, opts_cell{:});
                            data.wave(ii).Hs = wave(ii).Hs;
                            data.wave(ii).Hs(f >= 4) = NaN;
                            data.wave(ii).Hs_flags = Hs_flags;

                            [Tmean_flags, f] = QC_TEST(wave(ii).ddatetime, wave(ii).Tmean, app.sr, app.TP_QCsettings, opts_cell{:});
                            data.wave(ii).Tmean = wave(ii).Tmean;
                            data.wave(ii).Tmean(f >= 4) = NaN;
                            data.wave(ii).Tmean_flags = Tmean_flags;

                            [Tp_flags, f] = QC_TEST(wave(ii).ddatetime, wave(ii).Tp, app.sr, app.TP_QCsettings, opts_cell{:});
                            data.wave(ii).Tp = wave(ii).Tp;
                            data.wave(ii).Tp(f >= 4) = NaN;
                            data.wave(ii).Tp_flags = Tp_flags;
    
                            % [Dir_flags, f] = QC_TEST(ddatetime, Dir, sr, CD_QCsettings, opts_cell{:});
                            [Dir_flags, f] = QC_TEST(wave(ii).ddatetime, wave(ii).Dir, app.sr, app.WVD_QCsettings, "Test1", "on", "Test4", "on", "Test5", "on", ...
                                "Test6", "off", "Test7", "off", "Test8", "off", "Test10", "off");
                            data.wave(ii).Dir = wave(ii).Dir;
                            data.wave(ii).Dir(f >= 4) = NaN;
                            data.wave(ii).Dir_flags = Dir_flags;

                            data.wave(ii).raw_time = wave(ii).raw_time;
                            [data.wave(ii).Time_flags] = Test1(wave(ii).raw_time, app.sr, app.WV_QCsettings.t1_suspect, app.WV_QCsettings.t1_fail);

                        end

                end

                close(d);
                if userCancelled
                    uialert(app.UIFigure, 'QC cancelled by user.', 'Cancelled');
                    return; % 直接結束 StartButtonPushed 函式
                end

                if isempty(fieldnames(data))
                    uialert(app.UIFigure, "Selected wrong QC item!", "Error");
                    return;
                else
                    outputpath = string(app.OutputfolderEditField_2.Value);
                    inputitem = string(app.QCItem);
                    station = string(app.StationnameEditField_2.Value);
                    % station = app.stationname;
                    fileName = "QCedData_" + station + "_" + inputitem + ".mat";
                    save(fullfile(outputpath, fileName), '-struct', 'data')
                    uialert(app.UIFigure, "Saved in: " + newline + fullfile(outputpath, fileName), 'Success!');
                end
                
                catch ME
                    errorMessage = sprintf('Coding error! \n\nReason: %s\n\nLocation:%s (Line: %d)', ...
                    ME.message, ME.stack(1).name, ME.stack(1).line);
                    
                    uialert(app.UIFigure, errorMessage, 'System error', 'Icon', 'error');
            end
        end

        % Button pushed function: PropertiesButton
        function PropertiesButtonPushed(app, event)
            try
                if app.QCItemDropDown.Value == "Current (Vel, Dir)"
                    % 1. 打包主程式當前的設定（第一次打開若為空，就傳空值 []）
                    if ~isempty(app.CU_QCsettings)
                        currentSettings = app.CU_QCsettings;
                    else
                        currentSettings = []; % 第一次打開，讓子視窗使用預設值
                    end
            
                    % 2. 關鍵！把當前設定傳進子視窗括號裡：
                    settingsWin = Properties_QC_curr; 
                    if ~isempty(currentSettings)
                        settingsWin.loadSettings(currentSettings);
                    end
            
                    % 3. 等待使用者按 Ok / Cancel
                    uiwait(settingsWin.QCpropertiesCurrentUIFigure);
            
                    % 4. 判斷使用者是否按了 Ok
                    if isvalid(settingsWin)
                        if ~isempty(settingsWin.Results)
                            % 拿到最新修改的設定值
                            res = settingsWin.Results;
                            app.sr = settingsWin.samplingrate;
            
                            % 存回主 App 的各個變數中
                            app.CU_QCsettings = res;
                            % app.sr = sr;
            
                            uialert(app.UIFigure, 'Updated properties!', 'Success');
                        end
                        
                        % 不論按 Ok 還是 Cancel，等待結束後都把子視窗銷毀釋放記憶體
                        delete(settingsWin); 
                    end
                end

                if app.QCItemDropDown.Value == "Wave (Hs, Tp, Tmean, Dir)"
                    % 1. 打包主程式當前的設定（第一次打開若為空，就傳空值 []）
                    if ~isempty(app.WV_QCsettings)
                        currentSettings = app.WV_QCsettings;
                    else
                        currentSettings = []; % 第一次打開，讓子視窗使用預設值
                    end
            
                    % 2. 關鍵！把當前設定傳進子視窗括號裡：
                    settingsWin = Properties_QC_wave; 
                    if ~isempty(currentSettings)
                        settingsWin.loadSettings(currentSettings);
                    end
            
                    % 3. 等待使用者按 Ok / Cancel
                    uiwait(settingsWin.QCpropertiesWaveUIFigure);
            
                    % 4. 判斷使用者是否按了 Ok
                    if isvalid(settingsWin)
                        if ~isempty(settingsWin.Results_WV) || ~isempty(settingsWin.Results_TP)
                            % 拿到最新修改的設定值
                            res_WV = settingsWin.Results_WV;
                            res_TP = settingsWin.Results_TP;
                            app.sr = settingsWin.samplingrate;
            
                            % 存回主 App 的各個變數中
                            app.WV_QCsettings = res_WV;
                            app.TP_QCsettings = res_TP;
                            % app.sr = sr;
            
                            uialert(app.UIFigure, 'Updated properties!', 'Success');
                        end
                        
                        % 不論按 Ok 還是 Cancel，等待結束後都把子視窗銷毀釋放記憶體
                        delete(settingsWin); 
                    end
                end
        
            catch ME
                errorMessage = sprintf('Coding error! \n\nReason: %s\n\nLocation:%s (Line: %d)', ...
                    ME.message, ME.stack(end).name, ME.stack(end).line);
                uialert(app.UIFigure, errorMessage, 'System error', 'Icon', 'error');
            end
        end
